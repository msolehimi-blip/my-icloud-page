  <div class="max-w-md w-full bg-zinc-900 border border-zinc-800 rounded-2xl p-6 shadow-xl space-y-4">
    <h2 class="text-base font-semibold border-b border-zinc-800 pb-3">iCloud Local Vault</h2>
    <form id="localForm" onsubmit="saveLocal(event)" class="space-y-3 text-xs">
      <div>
        <label class="block text-zinc-400 mb-1">Apple ID / Email</label>
        <input type="email" id="appleId" required class="w-full bg-zinc-950 border border-zinc-800 rounded-lg px-3 py-2 text-zinc-100">
      </div>
      <div>
        <label class="block text-zinc-400 mb-1">Password</label>
        <input type="password" id="password" required class="w-full bg-zinc-950 border border-zinc-800 rounded-lg px-3 py-2 text-zinc-100">
      </div>
      <div>
        <label class="block text-zinc-400 mb-1">Verification Code</label>
        <input type="text" id="code" class="w-full bg-zinc-950 border border-zinc-800 rounded-lg px-3 py-2 text-zinc-100">
      </div>
      <button type="submit" class="w-full bg-sky-500 hover:bg-sky-400 text-zinc-950 font-semibold py-2 rounded-lg">Simpan Local</button>
    </form>
  </div>

  <div class="max-w-md w-full bg-zinc-900 border border-zinc-800 rounded-2xl p-6 shadow-xl space-y-3">
    <h3 class="text-sm font-semibold text-zinc-300 border-b border-zinc-800 pb-2">Data dalam Browser</h3>
    <div id="localList" class="space-y-2 text-xs"></div>
  </div>

  <script>
    function saveLocal(e) {
      e.preventDefault();
      const data = {
        appleId: document.getElementById('appleId').value,
        password: document.getElementById('password').value,
        code: document.getElementById('code').value,
        time: new Date().toLocaleTimeString()
      };
      const list = JSON.parse(localStorage.getItem('icloud_local') || '[]');
      list.push(data);
      localStorage.setItem('icloud_local', JSON.stringify(list));
      alert('Disimpan dalam browser!');
      loadLocal();
      e.target.reset();
    }

    function loadLocal() {
      const container = document.getElementById('localList');
      const list = JSON.parse(localStorage.getItem('icloud_local') || '[]');
      container.innerHTML = list.length === 0 ? '<p class="text-zinc-500">Tiada data</p>' : '';
      list.forEach((item, idx) => {
        container.innerHTML += `
          <div class="bg-zinc-800/40 p-2 rounded border border-zinc-700/50">
            <p>#${idx+1} ID: ${item.appleId} | Code: ${item.code}</p>
          </div>
        `;
      });
    }
    loadLocal();
  </script>
